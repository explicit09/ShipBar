# ShipBar V2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete local-first ShipBar V2 shipping cockpit across macOS and iPhone: daily Top 3 focus, truthful agent-run tracking and review, command-first navigation, upgraded triage and projects, premium native presentation, and end-to-end runtime proof.

**Architecture:** Extend the shared SwiftData schema with optional focus fields on `ShipTask` and a snapshot-based `AgentRun`. Keep all focus and run state transitions in pure coordinators under `Sources/Shared/Tasks`, then build query-driven SwiftUI surfaces on top. Existing capture, persistence, CloudKit fallback, project lifecycle, and agent-launch paths remain authoritative and are extended rather than duplicated.

**Tech Stack:** Swift 6, SwiftUI, SwiftData, AppKit menu-bar shell, iOS 17, macOS 14, Swift Testing, XcodeGen, CloudKit private database with local fallback.

## Global Constraints

- Preserve all existing task, project, capture, voice, share-extension, deletion, and CloudKit behavior.
- Never imply remote agent telemetry that ShipBar did not observe or the user did not report.
- Existing V1 stores must migrate without data loss; all new CloudKit-persisted properties have defaults or are optional.
- The Mac interface must remain useful at 420×620 and support full keyboard operation.
- iPhone is a focused companion and labels unsupported launches as “Ready to hand off on Mac.”
- No new third-party dependencies.
- Each task starts with a failing focused test, ends with a passing full test/build gate, and commits only its own files.
- Pre-existing uncommitted changes in the checkout are user-owned and must not be discarded or swept into V2 commits unintentionally.

```swift
import Foundation

enum FocusCoordinator {
    static let maximumCount = 3

    @discardableResult
    static func setFocus(_ task: ShipTask, among tasks: [ShipTask], on day: Date, calendar: Calendar = .current) -> Bool {
        normalize(tasks, on: day, calendar: calendar)
        if calendar.isDate(task.focusDate ?? .distantPast, inSameDayAs: day) { return true }
        let focused = focusedTasks(in: tasks, on: day, calendar: calendar)
        guard focused.count < maximumCount, task.status != .done else { return false }
        task.focusDate = calendar.startOfDay(for: day)
        task.focusOrder = focused.count + 1
        task.updatedAt = .now
        return true
    }

    static func removeFocus(_ task: ShipTask, among tasks: [ShipTask]) {
        task.focusDate = nil
        task.focusOrder = nil
        normalize(tasks, on: .now)
    }

    static func moveFocus(_ task: ShipTask, to position: Int, among tasks: [ShipTask], on day: Date, calendar: Calendar = .current) {
        var ordered = focusedTasks(in: tasks, on: day, calendar: calendar).filter { $0.id != task.id }
        ordered.insert(task, at: min(max(position - 1, 0), ordered.count))
        for (index, item) in ordered.enumerated() {
            item.focusDate = calendar.startOfDay(for: day)
            item.focusOrder = index + 1
        }
    }

    static func normalize(_ tasks: [ShipTask], on day: Date, calendar: Calendar = .current) {
        let ordered = focusedTasks(in: tasks, on: day, calendar: calendar).prefix(maximumCount)
        for (index, task) in ordered.enumerated() { task.focusOrder = index + 1 }
    }

    static func focusedTasks(in tasks: [ShipTask], on day: Date, calendar: Calendar = .current) -> [ShipTask] {
        tasks.filter { task in
            guard let date = task.focusDate else { return false }
            return calendar.isDate(date, inSameDayAs: day)
        }.sorted { ($0.focusOrder ?? .max, $0.createdAt) < ($1.focusOrder ?? .max, $1.createdAt) }
    }
}
```

- [ ] **Step 4: Implement explicit transitions and review decisions**

```swift
enum AgentRunLifecycle {
    private static let allowed: [AgentRunStatus: Set<AgentRunStatus>] = [
        .prepared: [.handedOff, .canceled, .failed],
        .handedOff: [.running, .needsReview, .failed, .canceled],
        .running: [.needsReview, .failed, .canceled],
        .needsReview: [.completed, .prepared, .failed, .canceled],
        .completed: [], .failed: [.prepared], .canceled: [.prepared],
    ]

    @MainActor static func prepare(task: ShipTask, target: AgentTarget, in context: ModelContext, now: Date = .now) -> AgentRun {
        let action = AgentWorkflowAction.make(for: target, task: task)
        let run = AgentRun(taskID: task.id, projectID: task.project?.id, taskTitleSnapshot: task.title,
            projectNameSnapshot: task.project?.name, targetRawValue: target.rawValue,
            statusRawValue: AgentRunStatus.prepared.rawValue, promptSnapshot: action.clipboardText,
            repositoryPathSnapshot: action.repoPath, createdAt: now, updatedAt: now)
        context.insert(run)
        return run
    }

    @discardableResult static func transition(_ run: AgentRun, to next: AgentRunStatus, now: Date = .now) -> Bool {
        guard allowed[run.status, default: []].contains(next) else { return false }
        run.status = next; run.updatedAt = now
        if next == .running { run.startedAt = run.startedAt ?? now }
        if [.completed, .failed, .canceled].contains(next) { run.finishedAt = now }
        return true
    }

    static func accept(_ run: AgentRun, task: ShipTask?, completeTask: Bool, now: Date = .now) {
        guard transition(run, to: .completed, now: now) else { return }
        if completeTask { task?.applyStatus(.done, now: now) }
    }
}
```

Implement `markHandedOff` through `transition`, and `requestChanges` by creating a new prepared run from the current task while leaving the reviewed run in history.
